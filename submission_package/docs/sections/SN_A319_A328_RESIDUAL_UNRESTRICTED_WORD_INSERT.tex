\section{SN-A319 --- Soft Characterization Theorem}
\label{sec:SN-A319}

\begin{definition}[Residual soft-penalty class $\mathcal{S}$]
\label{def:A319-S}
A map $s:\{\text{even }n\ge 2\}\to[0,\infty)$ lies in $\mathcal{S}$ if:
\begin{enumerate}[label=(S\arabic*)]
\item $s(6)=0$;
\item reflection symmetry about packing Maxwell: $s(n)=s(12-n)$ for even $n\in\{2,\ldots,10\}$;
\item $g(k):=s(2k)$ is discrete-convex on integers $k\ge 1$;
\item unit residual slope: $g(4)-g(3)=g(2)-g(3)=1$;
\item free continuous primary parameters $=0$ (no free overall scale beyond the unit slope).
\end{enumerate}
\end{definition}

\begin{theorem}[Soft characterization]
\label{thm:A319}
The unique element of $\mathcal{S}$ is
\begin{equation}

\end{equation}
In particular $\mathrm{soft}(6)=0$, $\mathrm{soft}(4)=\mathrm{soft}(8)=1$, $\mathrm{soft}(10)=2$, $\mathrm{soft}(12)=3$.
\end{theorem}

\begin{proof}
Write $g(k)=s(2k)$. By (S1), $g(3)=0$. By (S4), $g(4)=1$ and $g(2)=1$. Discrete convexity (S3) forces the right difference sequence to be non-decreasing: $g(k+1)-g(k)\ge g(k)-g(k-1)$. Starting from unit right slope at $k=3$, the minimal convex extension is $g(3+t)=t$ for $t\ge 0$. Reflection (S2) gives $g(3-t)=t$. Hence $g(k)=|k-3|$, i.e.\ $s(n)=|n/2-3|$.

Uniqueness: any other convex $g$ with the same values at $\{2,3,4\}$ and non-decreasing slopes coincides with $|k-3|$ on $\mathbb{Z}$. Free continuous rescaling $c|k-3|$ with $c\neq 1$ violates (S4) or (S5). Squared penalty $(k-3)^2$ has vanishing derivative at the minimum, violating (S4). Certificate: \texttt{A319.json}.
\end{proof}

\end{remark}
